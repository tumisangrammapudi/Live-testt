// This Service Worker forces an update by generating a unique CACHE_NAME on 
every load.
const CACHE_NAME = `dependent-app-cache-${Date.now()}`;
const urlsToCache = [
 './',
 './index.html',
 './manifest.webmanifest',
 // Note: We removed specific icon paths since they were failing before.
 // Add icon paths here ONLY if they exist and are correctly named.
];

// Install Event: Caches necessary assets
self.addEventListener('install', event => {
 event.waitUntil(
 caches.open(CACHE_NAME)
 .then(cache => {
 console.log('SW: Opening cache and adding total assets:', urlsToCache.length);
 // Using addAll might fail if any single URL fails. Use with caution.
 return cache.addAll(urlsToCache);
 })
 // Force the waiting service worker to become active immediately
 .then(() => self.skipWaiting())
 );
});

// Activate Event: Cleans up old caches
self.addEventListener('activate', event => {
 event.waitUntil(
 caches.keys().then(cacheNames => {
 return Promise.all(
 cacheNames.map(cache => {
 if (cache !== CACHE_NAME) {
 console.log('SW: Deleting old cache:', cache);
 return caches.delete(cache);
 }
 })
 );
 })
 // Immediately take control of all open clients/tabs
 .then(() => self.clients.claim())
 );
});

// Fetch Event: network-first, then cache fallback
self.addEventListener('fetch', event => {
 event.respondWith(
 fetch(event.request)
 .then(networkResponse => {
 // If successful network request, return it
 if (networkResponse && networkResponse.status === 200) {
 return networkResponse;
 }
 // If network response failed, try cache
 return caches.match(event.request);
 })
 .catch(() => {
 // If network request failed completely (offline), try cache
 return caches.match(event.request);
 })
 );
});
